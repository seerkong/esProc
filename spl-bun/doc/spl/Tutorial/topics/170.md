# 6.4.2 配置 raqsoftConfig.xml

## 内容

### 6.4.2 配置 raqsoftConfig.xml

raqsoftConfig.xml 文件包含配置信息，包括 esProc 的主路径、脚本文件的搜索路径等，如下所示：

```xml
<?xml version="1.0" encoding="UTF-8"?>
<Config Version="3">
  <Runtime>
    <!-- 数据源配置方法一：配置连接池并在应用服务器中指定数据源名称 -->
    <!-- encryptLevel 是数据源中用户密码加密的级别：0 表示明文，1 表示密码加密，2 表示 URL、用户名和密码加密 -->
    <DBList encryptLevel="0">
      <!-- 数据源名称必须与脚本文件中定义的名称相同 -->
      <DB name="demo">
        <property name="url" value="jdbc:hsqldb:hsql://127.0.0.1/demo"/>
        <property name="driver" value="org.hsqldb.jdbcDriver"/>
        <!-- type 表示数据库类型的序号：UNKNOWN:0; ORACLE:1; SQLSVR:2; SYBASE:3; SQLANY:4; INFMIX:5; FOXPRO:6; ACCESS:7; FOXBAS:8; DB2:9; MYSQL:10; KINGBASE:11; DERBY:12; HSQL:13; TERADATA:14; POSTGRES:15; DATALOGIC:16; IMPALA:17; DBOne:20; ESSBASE:101; -->
        <property name="type" value="13"/>
        <property name="user" value="sa"/>
        <property name="password" value=""/>
        <!-- 设置批处理大小 -->
        <property name="batchSize" value="1000"/>
        <!-- 是否自动连接。如果值为 true，直接使用 db.query() 函数或以 $ 开头的 SQL 语句访问数据库；如果为 false，则不会自动连接；您应该先使用 connect(db) 语句建立连接，然后才能访问数据库。 -->
        <property name="autoConnect" value="true"/>
        <!-- 设置是否使用 schema 名称 -->
        <property name="useSchema" value="false"/>
        <!-- 设置是否用引号括起对象 -->
        <property name="addTilde" value="false"/>
        <!-- 设置是否对从数据库检索的内容进行转码 -->
        <property name="needTransContent" value="false"/>
        <!-- 设置是否对 SQL 语句进行转码 -->
        <property name="needTransSentence" value="false"/>
        <!-- 设置 SQL 语句是否区分大小写 -->
        <property name="caseSensitive" value="false"/>
      </DB>
    </DBList>
    <esProc>
      <!-- esProc 字符集 -->
      <charSet>ISO-8859-1</charSet>
      <!-- 配置脚本文件的搜索路径。将多个路径分别写在 <splPath></splPath> 节点中。脚本文件也可以放在应用程序项目的类路径中，类路径在文件加载时优先于搜索路径。 -->
      <splPathList>
        <splPath>D:\files\spl</splPath>
        <splPath>D:\files\txt</splPath>
        <splPath>D:\files</splPath>
      </splPathList>
      <!-- 日期类型、时间类型和其他数据类型 -->
      <dateFormat>yyyy-MM-dd</dateFormat>
      <timeFormat>HH:mm:ss</timeFormat>
      <dateTimeFormat>yyyy-MM-dd HH:mm:ss</dateTimeFormat>
      <!-- esProc 主路径，是单个绝对路径 -->
      <mainPath>D:\files</mainPath>
      <!-- 临时文件的保存路径 -->
      <tempPath>temp</tempPath>
      <!-- 设置文件缓冲区大小，单位为字节 -->
      <bufSize>65536</bufSize>
      <!-- 设置最大并行任务数 -->
      <parallelNum>4</parallelNum>
      <!-- 设置组合文件块大小 -->
      <blockSize>1048576</blockSize>
      <!-- 设置多游标中子游标的默认数量 -->
      <cursorParallelNum>1</cursorParallelNum>
      <!-- 文本数据中的空字符串 -->
      <nullStrings>nan,null,n/a</nullStrings>
      <!-- 设置从游标获取的记录数 -->
      <fetchCount>9999</fetchCount>
      <!-- 配置外部库目录 -->
      <extLibsPath/>
    </esProc>
    <Logger>
      <!-- 日志级别，可以是 OFF、SEVERE、WARNING、INFO、DEBUG；它们的优先级按顺序递减 -->
      <Level>DEBUG</Level>
    </Logger>
  </Runtime>
  <Server>
    <!-- 默认连接；连接的源可以是 DBList 中的一个或以下 JNDIList 中的一个 -->
    <defDataSource>demo</defDataSource>
    <!-- 设置 JNDI 数据源 -->
    <JNDIList>
      <JNDI name="jdbc/test">
        <property name="type" value="1"/>
        <property name="batchSize" value="1000"/>
        <property name="lookup" value="java:comp/env/jdbc/test"/>
      </JNDI>
    </JNDIList>
    <!-- 日志记录配置属性文件 -->
    <property name="logConfig" value="raqsoftLog.properties"/>
  </Server>
  <JDBC>
    <!-- 是否读取运行时配置或 Server 的 JNDIList -->
    <load>Runtime,Server</load>
    <!-- 配置 jdbc 网关后，所有语句将由脚本文件解析并返回序表或游标 -->
    <gateway>executeSQL.splx</gateway>
    <!-- 用于 JDBC 调用的远程服务器列表 -->
    <Units>
      <Node>192.168.0.197:8281</Node>
    </Units>
  </JDBC>
  <Init>
    <!-- 系统初始化时首先执行的指定脚本文件 -->
    <SPL>initial.spl</SPL>
  </Init>
</Config>
```

| 服务器 | Tomcat | Weblogic | Websphere | Resin | Jboss |
|--------|--------|----------|-----------|-------|-------|
| JNDI前缀 | java:comp/env | 无 | 无 | java:comp/env | Java: |

在数据源管理器中，您可以设置数据源驱动程序、数据源 URL、用户名、密码、批处理大小和其他信息。请注意，demo 是系统数据源，其连接参数无法更改。此外，在数据源连接配置期间，esProc JDBC 本身不能用作数据源。

`<Server/>` 节点配置连接池、日志配置文件和其他信息。如果通过 esProc JDBC 调用的脚本文件需要连接数据源，相关信息应在 raqsoftConfig.xml 的数据源或连接池中配置。特别是，如果使用 db.query() 函数或 $(db) sql 语句直接访问数据库，此数据库的 autoConnect 属性应为 true。有关 esProc JDBC 的详细信息，请参阅"Java 调用"。

在日志配置属性文件 raqsoftLog.properties 中，您可以配置日志级别和其他信息：

```properties
// 日志级别包括 OFF、SEVERE、WARN、INFO、DEBUG，其优先级从左到右递减。级别 OFF 关闭日志记录。
// 指定 Logger 的名称和日志级别。
Logger=LOG1,LOG2
// 将日志消息输出到控制台。
LOG1=Console
LOG1.level=DEBUG
// 将日志输出到指定文件。
LOG2=C:/raqsoft.log
// 日志文件中的最大字节数；默认值为 10M。
LOG2.maxSize=10M
// 日志文件编码；默认为 UTF-8。
LOG2.encoding=GBK
// 是否使日志名称固定。
LOG2.isFixedFileName=false
```

在使用 esProc JDBC 时，Java 应用程序类路径中的 raqsoftConfig.xml 文件将自动加载，此时不能重命名。但如果 esProc JDBC 的连接字符串使用类似 ...?config=... 的格式，则将加载连接字符串中 config 指定的 .xml 文件用于配置，并忽略配置中定义的默认文件。

例如，使用类似 `con = DriverManager.getConnection("jdbc:esproc:local:/?config=myconfig.xml")` 的连接字符串，将使用 myconfig.xml 中的配置。
